System Design

Deterministic Single-Pass Architecture

Built to avoid decode queues, memory bloat, and false positives on mobile browsers without requiring Cross-Origin Isolation (COOP/COEP).

  1. 01

    CameraAdapter

    Requests rear stream (≥1080p primary, 720p fallback) and auto-pauses on visibilitychange.

  2. 02

    LatestFrameScheduler

    Enforces queueDepth ≤ 1, drops stale buffers, and rejects JPEG/base64 payloads (ADR 0002).

  3. 03

    Native / WASM Decoder

    Prefers native BarcodeDetector when supported; seamlessly runs local zxing-wasm@3.1.7.

  4. 04

    GS1 Checksum & Canonicalizer

    Validates mod-10 check digit and normalizes 12-digit UPC-A to 13-digit EAN-13 (0 prefix).

  5. 05

    2-Frame ConsensusTracker

    Requires 2 agreeing checksum-valid observations within 1500 ms with 2000 ms duplicate cooldown.

  6. 06

    ScanResult / Manual Fallback

    Emits structured evidence; universal manual entry stays reachable in 100% of error states.

Integration Guide

Embed the turnkey accessible DOM view, bring your own React 19 UI, or isolate WASM decoding inside a dedicated Web Worker. 

import { WebBarcodeScanner } from "open-retail-scanner-web";

const root = document.getElementById("scanner-root");
if (!root) throw new Error("Missing #scanner-root container");

const scanner = new WebBarcodeScanner({
  formats: ["ean13", "upca", "ean8"],
  requiredConfirmingFrames: 2,
  consensusWindowMs: 1500,
  duplicateCooldownMs: 2000,
  preferNativeDetector: true,
  zxingOptions: {
    wasmUrl: "/wasm/zxing_reader.wasm",
  },
  onEvent(event) {
    if (event.type === "result") {
      // Pass canonicalValue in memory to your app's product lookup.
      // Never log rawValue or canonicalValue to console or telemetry.
      void scanner.stop();
      handleConfirmedBarcode(event.result);
    }
  },
});

const view = scanner.mount(root);
await scanner.start();

import { useEffect, useRef, useState } from "react";
import {
  WebBarcodeScanner,
  getAccessibleErrorMessage,
  type ScannerUIState,
} from "open-retail-scanner-web";
import type { ScanResult } from "open-retail-scanner-core";

export function ProductBarcodeScanner({
  wasmUrl = "/wasm/zxing_reader.wasm",
  onBarcodeConfirmed,
}: {
  wasmUrl?: string;
  onBarcodeConfirmed: (result: ScanResult) => void;
}) {
  const videoRef = useRef<HTMLVideoElement | null>(null);
  const scannerRef = useRef<WebBarcodeScanner | null>(null);
  const [uiState, setUiState] = useState<ScannerUIState | null>(null);

  useEffect(() => {
    const scanner = new WebBarcodeScanner({
      formats: ["ean13", "upca", "ean8"],
      preferNativeDetector: true,
      zxingOptions: { wasmUrl },
      onEvent(event) {
        if (event.type === "result") {
          void scanner.stop();
          onBarcodeConfirmed(event.result);
        }
      },
    });
    scannerRef.current = scanner;
    const unsub = scanner.getUIController().subscribe(setUiState);
    if (videoRef.current) scanner.attachVideoElement(videoRef.current);
    void scanner.start();

    return () => {
      unsub();
      scanner.attachVideoElement(null);
      void scanner.dispose();
    };
  }, [wasmUrl, onBarcodeConfirmed]);

  return (
    <section aria-label="Barcode scanner">
      <video ref={videoRef} playsInline muted autoPlay />
      {uiState?.state === "error" && uiState.errorCode && (
        <p role="alert">
          {getAccessibleErrorMessage(uiState.errorCode, uiState.errorRecoverable)}
        </p>
      )}
    </section>
  );
}

import {
  ZxingWasmAdapter,
  attachDecoderWorkerScope,
  type WorkerGlobalScopeLike,
} from "open-retail-scanner-web";

const adapter = new ZxingWasmAdapter({
  formats: ["ean8", "ean13", "upca"],
  wasmUrl: "/wasm/zxing_reader.wasm",
});

attachDecoderWorkerScope(
  self as unknown as WorkerGlobalScopeLike,
  async (frame) => {
    await adapter.init();
    return adapter.decodeFrame(frame);
  },
);

// 2. Main thread scanner setup
import {
  WebBarcodeScanner,
  createWebWorkerTransport,
  type WebWorkerLike,
} from "open-retail-scanner-web";

const scanner = new WebBarcodeScanner({
  zxingOptions: { wasmUrl: "/wasm/zxing_reader.wasm" },
  createWorkerTransport: () => {
    const worker = new Worker(new URL("./scanner.worker.ts", import.meta.url), {
      type: "module",
    });
    return createWebWorkerTransport(worker as unknown as WebWorkerLike, {
      transferBuffer: true,
    });
  },
});
