[ 03 // DETERMINISTIC PIPELINE ]

Single-Pass Optical Architecture

Engineered to eliminate decode queues, memory spikes, and false reads on mobile browsers without requiring Cross-Origin Isolation (COOP/COEP).

  1. STAGE // 01

    CameraAdapter

    Acquires rear video stream (≥1080p primary, 720p fallback) and auto-pauses on visibilitychange.

  2. STAGE // 02

    LatestFrameScheduler

    Enforces queueDepth ≤ 1, drops stale buffers immediately, and rejects JPEG/base64 payloads (ADR 0002).

  3. STAGE // 03

    Native / WASM Decoder

    Prefers native BarcodeDetector when supported; seamlessly executes local zxing-wasm@3.1.7.

  4. STAGE // 04

    GS1 Checksum & Canonicalizer

    Validates mod-10 check digit and normalizes 12-digit UPC-A into 13-digit EAN-13 (leading 0).

  5. STAGE // 05

    2-Frame ConsensusTracker

    Requires 2 matching checksum-valid reads within 1500 ms and applies 2000 ms duplicate cooldown.

  6. STAGE // 06

    ScanResult / Manual Fallback

    Emits structured evidence; universal manual entry remains reachable in 100% of error and lifecycle states.

import { useEffect, useRef, useState } from "react";
import {
  WebBarcodeScanner,
  getAccessibleErrorMessage,
  type ScannerUIState,
} from "open-retail-scanner-web";
import type { ScanResult } from "open-retail-scanner-core";

export function ProductBarcodeScanner({
  wasmUrl = "/wasm/zxing_reader.wasm",
  onBarcodeConfirmed,
}: {
  wasmUrl?: string;
  onBarcodeConfirmed: (result: ScanResult) => void;
}) {
  const videoRef = useRef<HTMLVideoElement | null>(null);
  const scannerRef = useRef<WebBarcodeScanner | null>(null);
  const [uiState, setUiState] = useState<ScannerUIState | null>(null);

  useEffect(() => {
    const scanner = new WebBarcodeScanner({
      formats: ["ean13", "upca", "ean8"],
      preferNativeDetector: true,
      zxingOptions: { wasmUrl },
      onEvent(event) {
        if (event.type === "result") {
          void scanner.stop();
          onBarcodeConfirmed(event.result);
        }
      },
    });
    scannerRef.current = scanner;
    const unsub = scanner.getUIController().subscribe(setUiState);
    if (videoRef.current) scanner.attachVideoElement(videoRef.current);
    void scanner.start();

    return () => {
      unsub();
      scanner.attachVideoElement(null);
      void scanner.dispose();
    };
  }, [wasmUrl, onBarcodeConfirmed]);

  return (
    <section aria-label="Barcode scanner">
      <video ref={videoRef} playsInline muted autoPlay />
      {uiState?.state === "error" && uiState.errorCode && (
        <p role="alert">
          {getAccessibleErrorMessage(uiState.errorCode, uiState.errorRecoverable)}
        </p>
      )}
    </section>
  );
}

import {
  ZxingWasmAdapter,
  attachDecoderWorkerScope,
  type WorkerGlobalScopeLike,
} from "open-retail-scanner-web";

const adapter = new ZxingWasmAdapter({
  formats: ["ean8", "ean13", "upca"],
  wasmUrl: "/wasm/zxing_reader.wasm",
});

attachDecoderWorkerScope(
  self as unknown as WorkerGlobalScopeLike,
  async (frame) => {
    await adapter.init();
    return adapter.decodeFrame(frame);
  },
);

  WebBarcodeScanner,
  createWebWorkerTransport,
  type WebWorkerLike,
} from "open-retail-scanner-web";

const scanner = new WebBarcodeScanner({
  zxingOptions: { wasmUrl: "/wasm/zxing_reader.wasm" },
  createWorkerTransport: () => {
    const worker = new Worker(new URL("./scanner.worker.ts", import.meta.url), {
      type: "module",
    });
    return createWebWorkerTransport(worker as unknown as WebWorkerLike, {
      transferBuffer: true,
    });
  },
});
